% Type scale (size / leading). Leading is set per language in cv*.tex via
% \SetLeading, because CJK text needs more line height than Latin text.

\newcommand{\SetLeading}[1]{\renewcommand{\BodyLeading}{#1}}
\newcommand{\BodyFont}{\fontsize{10}{\BodyLeading}\selectfont}
\newcommand{\MetaFont}{\fontsize{9.3}{\BodyLeading}\selectfont\color{muted}}
\newcommand{\EntryFont}{\fontsize{10.3}{\BodyLeading}\selectfont\bfseries\color{ink}}

\newcommand{\TaglineFont}{\fontsize{10.5}{13}\selectfont\color{muted}}
\newcommand{\ContactFont}{\fontsize{9.3}{12}\selectfont\color{muted}}

% Section-title casing hook: English uppercases with tracking, Chinese keeps
% the text as is. Set in cv_en.tex / cv.tex.
\newcommand{\SectionText}[1]{#1}

% CJK: Source Han Sans / Noto Sans CJK is the designed companion of
% Source Sans. CI installs fonts-noto-cjk; other names are local fallbacks.
\newcommand{\UseCJKFont}[2]{%
  \setCJKmainfont{#1}[BoldFont = {#2}]%
  \setCJKsansfont{#1}[BoldFont = {#2}]%
}
\IfFontExistsTF{Noto Sans CJK SC}{\UseCJKFont{Noto Sans CJK SC}{Noto Sans CJK SC Bold}}{%
\IfFontExistsTF{Source Han Sans SC}{\UseCJKFont{Source Han Sans SC}{Source Han Sans SC Bold}}{%
\IfFontExistsTF{Hiragino Sans GB}{\UseCJKFont{Hiragino Sans GB}{Hiragino Sans GB W6}}{%
  \setCJKmainfont{FandolHei-Regular.otf}[BoldFont = FandolHei-Bold.otf]%
  \setCJKsansfont{FandolHei-Regular.otf}[BoldFont = FandolHei-Bold.otf]%
}}}

% \Header{Name}{Tagline}{\Contact{...}\Contact{...}...}[optional photo path]
% Separators between \Contact items are added automatically and vanish at a
% line break; an item itself is never broken across lines.
\newcommand{\Sep}{\hspace{0.55em}{\color{rulecolor}|}\hspace{0.55em}}
\newcommand{\ContactSep}{}
\newcommand{\Contact}[1]{\ContactSep\mbox{#1}\gdef\ContactSep{\discretionary{}{}{\mbox{\Sep}}}}
\NewDocumentCommand{\Header}{m m m o}{%
  \IfNoValueTF{#4}{%
    \HeaderText{#1}{#2}{#3}%
  }{%
    \noindent
    \begin{minipage}[b]{\dimexpr\linewidth-2.6cm\relax}\HeaderText{#1}{#2}{#3}\end{minipage}%
    \hfill
    \begin{minipage}[b]{2.05cm}\includegraphics[width=2.05cm]{#4}\end{minipage}\par
  }%
  \vspace{2pt}%
}
\newcommand{\HeaderText}[3]{%
  \raggedright
  {\NameFont #1\par}\vspace{3pt}%
  {\TaglineFont #2\par}\vspace{5pt}%
  {\ContactFont #3\par}%
}

% \Section{Title}
\newcommand{\Section}[1]{%
  \par\addvspace{\SectionSkip}%
  \needspace{4\baselineskip}%
  {\SectionFont\SectionText{#1}\par}%
  \nobreak\vspace{2.5pt}%
  {\color{rulecolor}\hrule height 0.6pt}%
  \nobreak\vspace{\SectionGap}%
  \BodyFont
}

% \Entry{Title}{Right-aligned meta, e.g. dates}
% The meta column has natural width, so long venues or dates never wrap
% into a fixed-width box.
\newcommand{\Entry}[2]{%
  \par\addvspace{\EntrySkip}%
  \needspace{3\baselineskip}%
  {\EntryFont #1}\hspace{1em plus 1fill}{\MetaFont #2}\par\nobreak
  \BodyFont
}

% \Sub{Secondary line under an entry: role, degree, group}
\newcommand{\Sub}[1]{{\fontsize{9.8}{\BodyLeading}\selectfont\color{muted}#1\par}\nobreak}

% \Note{Plain paragraph inside an entry}
\newcommand{\Note}[1]{{\BodyFont #1\par}}

% \Item{Bullet text}
\newcommand{\Item}[1]{%
  \par\vspace{\ItemSkip}%
  {\BodyFont\hangindent=\BulletIndent\hangafter=1
   \makebox[\BulletIndent][l]{\color{muted}\textbullet}#1\par}%
}

% \Pub{Title}{Authors}{Venue}{Links}; wrap your own name in \Me{...}
\newcommand{\Me}[1]{{\color{ink}\bfseries #1}}
\newcommand{\Pub}[4]{%
  \par\addvspace{\EntrySkip}%
  \needspace{2\baselineskip}%
  {\fontsize{10}{\BodyLeading}\selectfont\bfseries\color{ink}#1}\hspace{1em plus 1fill}{\MetaFont #3}\par\nobreak
  {\MetaFont #2\ifx\relax#4\relax\else\hspace{0.5em}#4\fi\par}%
}

% \Skill{Label}{Items}
\newcommand{\Skill}[2]{%
  \par\vspace{\ItemSkip}%
  {\BodyFont{\bfseries #1}\hspace{0.6em}#2\par}%
}

% \Award{Award text}{Year}
\newcommand{\Award}[2]{%
  \par\vspace{\ItemSkip}%

}
